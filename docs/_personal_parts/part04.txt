function inviteGraphSvg(edges, meId){
 if(!edges||!edges.length){ return '<svg viewBox="0 0 320 56"><text x="160" y="32" text-anchor="middle" fill="#B3A893" font-size="13">Граф появится с первым приглашением</text></svg>'; }
 var nodes={}, adj={}, indeg={};
 edges.forEach(function(e){ if(!e.from||!e.to) return; nodes[e.from]=1; nodes[e.to]=1; (adj[e.from]=adj[e.from]||[]).push(e.to); indeg[e.to]=(indeg[e.to]||0)+1; });
 var ids=Object.keys(nodes); if(!ids.length) return '';
 var depth={}, roots=ids.filter(function(id){ return !indeg[id]; }); var q=roots.slice(); roots.forEach(function(r){depth[r]=0;});
 while(q.length){ var u=q.shift(); (adj[u]||[]).forEach(function(v){ if(depth[v]==null){ depth[v]=(depth[u]||0)+1; q.push(v); } }); }
 ids.forEach(function(id){ if(depth[id]==null) depth[id]=0; });
 var byD={}; ids.forEach(function(id){ (byD[depth[id]]=byD[depth[id]]||[]).push(id); });
 var maxD=0; ids.forEach(function(id){ if(depth[id]>maxD) maxD=depth[id]; });
 var W=320, rowH=56, padY=24, pos={};
 for(var d=0; d<=maxD; d++){ var row=byD[d]||[]; row.forEach(function(id,i){ pos[id]={ x: W*(i+1)/(row.length+1), y: padY + d*rowH }; }); }
 var Hh = padY*2 + maxD*rowH;
 var E=edges.map(function(e){ var a=pos[e.from], b=pos[e.to]; if(!a||!b) return ''; return '<line x1="'+a.x.toFixed(1)+'" y1="'+a.y.toFixed(1)+'" x2="'+b.x.toFixed(1)+'" y2="'+b.y.toFixed(1)+'" stroke="#3a3329" stroke-width="1.2"/>'; }).join('');
 var N=ids.map(function(id){ var p=pos[id]; var me=(meId&&id===meId); return '<circle cx="'+p.x.toFixed(1)+'" cy="'+p.y.toFixed(1)+'" r="'+(me?7:5)+'" fill="'+(me?'#E7C07A':'#6E665A')+'"'+(me?' stroke="#fff" stroke-width="1.5"':'')+'/>'; }).join('');
 return '<svg viewBox="0 0 '+W+' '+Hh+'">'+E+N+'</svg>';
}
